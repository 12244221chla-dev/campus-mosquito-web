四、系统功能模块
模块 1：隐患点位管理（顺序表）
存储校园 7 个灭蚊片区信息。
字段：点位编号、片区名称、风险等级、整改状态、隐患备注。
功能：新增片区、修改片区信息、删除片区、浏览全部片区。
整改状态：0 = 未整改，1 = 已整改
模块 2：排查日志管理（单链表）
每次现场排查消杀之后，录入一条日志。
字段：片区编号、排查时间、隐患描述、消杀处理方式、本次整改结果。
功能：新增排查记录、查询某片区全部历史排查日志。
模块 3：任务优先级队列（核心功能）
生成每日排查任务清单。
优先级规则（从高到低）：
高危 + 未整改（优先处理）
中危 + 未整改
已整改区域（最后安排常规复查）
模块 4：排序与二分查找
排序：所有片区按风险等级、整改状态排序输出。
二分查找：输入片区编号，快速定位该片区，查看隐患信息。
模块 5：隐患数据统计
统计输出：
高危片区总数、中危片区总数
未整改隐患片区数量
已完成整改片区数量
方便后勤人员直观掌握校园蚊虫隐患分布。

#define MAXSITE 20    //顺序表最大片区数量，7个片区足够用

// 隐患点位（顺序表存储）
typedef struct{
    int id;             //片区编号
    char name[60];      //片区名称
    int risk;           //风险等级：1高危，2中危
    int status;         //0未整改，1已整改
    char note[120];     //隐患备注
} Site;

// 顺序表结构
typedef struct{
    Site data[MAXSITE];
    int length;
} SiteList;

// 排查日志，单链表节点
typedef struct LogNode{
    int siteId;         //对应片区编号
    char time[25];      //排查时间
    char problem[120];  //隐患描述
    char handle[120];   //消杀处理方式
    int result;         //0未完成整改，1完成整改
    struct LogNode *next;
} LogNode, *LogList;

//优先队列任务节点
typedef struct{
    int siteId;
    int risk;
    int status;
} Task;
